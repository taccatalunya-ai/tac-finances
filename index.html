  async function afegirFestivalRapid(tipus){

    if(!isAdmin()){ toast("No tens permisos", true); return; }

    const nom = prompt("Nom del festival a afegir:");

    if(!nom || !nom.trim()) return;

    

    const existents = FESTIVALS.filter(f => f.year === currentYear && f.tipus === tipus);

    const maxOrdre = existents.length > 0 ? Math.max(...existents.map(f=>f.ordre)) : -1;


    const { data, error } = await sb.from("festivals").insert({

      year: currentYear, 

      nom: nom.trim(), 

      tipus: tipus,

      ordre: maxOrdre + 1,

      quota_import: tipus === 'tac' ? QUOTA_TAC : QUOTA_AMICS,

      quota_pagada: false,

      estat: 'pendent'

    }).select().single();

    

    if(error){ toast("Error afegint: " + error.message, true); console.error(error); return; }

    FESTIVALS.push(data);

    toast("Afegit"); renderMain();

  }

    if(!isAdmin()){ toast("No tens permisos", true); return; }

    if(!confirm("Confirma-ho una altra vegada: esborrar l'any " + any + " és irreversible.")) return;

    try{

      // Esborrar fitxers d'actes del Storage primer

      const actesAny = ACTES.filter(a => Number(a.year) === Number(any));

      if(actesAny.length > 0){

        const paths = actesAny.map(a => a.fitxer_path);

        await sb.storage.from('actes').remove(paths);

      }


      const { error: e1 } = await sb.from("finances").delete().eq("year", any);

      if(e1) throw e1;

      const { error: e2 } = await sb.from("festivals").delete().eq("year", any);

      if(e2) throw e2;

      const { error: e3 } = await sb.from("actes").delete().eq("year", any);

      if(e3) throw e3;


      YEARS = YEARS.filter(y => y !== any);

      if(DATA && DATA[any]) delete DATA[any];

      FESTIVALS = FESTIVALS.filter(f => Number(f.year) !== any);

    if(nouIndex < 0 || nouIndex >= mateixTipus.length) return;


    const temp = mateixTipus[index].ordre;

    mateixTipus[index].ordre = mateixTipus[nouIndex].ordre;

    mateixTipus[nouIndex].ordre = temp;


    const { error: e1 } = await sb.from("festivals").update({ordre: mateixTipus[index].ordre}).eq("id", mateixTipus[index].id);

    const { error: e2 } = await sb.from("festivals").update({ordre: mateixTipus[nouIndex].ordre}).eq("id", mateixTipus[nouIndex].id);

    

    if(e1 || e2) { toast("Error movent", true); console.error(e1, e2); return; }

    renderMain();

  }

      ev.preventDefault();

        const existents = FESTIVALS.filter(x => x.year === currentYear && x.tipus === selTipus.value && x.id !== f.id);

        ordreFinal = existents.length > 0 ? Math.max(...existents.map(x=>x.ordre)) + 1 : 0;

        tipus: selTipus.value,

        data_inici: ddmmaaaaToISO(dIni.value),

        data_fi: ddmmaaaaToISO(dFi.value),

        quota_import: quotaImport,

        estat: selEstat.value,

        notes: taNotes.value.trim() || null,

        updated_at: new Date().toISOString()

      };

      if(isEdit){

        btnSave.disabled = false; btnSave.textContent = "Desar canvis";

        if(error){ toast("Error afegint: " + error.message, true); console.error(error); return; }

        toast("Festival afegit");

      }


      await fetchFestivals();

      renderMain();

    });


    panel.appendChild(form);

    return panel;

  }

    if(!isAdmin()){ toast("No tens permisos", true); return; }

    toast("Eliminat"); renderMain();

  }
